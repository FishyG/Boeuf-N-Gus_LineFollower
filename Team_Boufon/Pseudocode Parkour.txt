// ---- Broches ----
PIN_VERT  ← 48          // detecteur, sortie verte
PIN_ROUGE ← 49          // detecteur, sortie rouge
PIN_5KHZ  ← A0          // circuit sifflet : signal 5 kHz
PIN_AMBIANT ← A1        // circuit sifflet : bruit ambiant
SEUIL_SIFFLET ← 50

// ---- Vitesse et PI ----
VITESSE ← 0,40
KP ← 0,002              
KI ← 0,001      
MAX_SOMME ← 2000    

// ---- Robot et parcours ----
TAILLE_CASE ← 0,5 m
ENTRAXE     ← 187 mm    // distance entre les 2 roues (a calibrer)
INFINI      ← 999

// ---- Grille : 10 rangees × 3 colonnes ----
//   rangee 9  [Fin][Fin][Fin]          y = 5,0 m
//     ...
//   rangee 0  [   ][Dep][   ]          y = 0 m
//            col 0  col 1  col 2
//          x=0   x=0,5  x=1,0  x=1,5
// cotes : HAUT (vers la fin), DROITE, BAS (vers le depart), GAUCHE


// ===================== VARIABLES =====================
// position_tape (reel)  : tableau des tapes [x, y] en metres
// position_robot (ent.) : case du robot [rangee, colonne]
// prochaine_pos (ent.)  : case où le robot veut aller [rangee, colonne]
// direction (ent.)      : cote que le robot regarde (HAUT, DROITE, BAS, GAUCHE)
// vert, rouge (bool.)   : sorties du detecteur
// mur[r][c][cote]       : VRAI = mur, panneau ou tape
// connu[r][c][cote]     : VRAI = cote deja verifie
// etat (ent.)           : ATTENTE, ALLER, RETOUR, TERMINE, ERREUR

FIN_CIBLES    ← { [9,0], [9,1], [9,2] }
DEPART_CIBLES ← { [0,1] }


DEBUT

// Chaque tape est note par le MILIEU de son segment, en metres :
//   tape vertical (entre 2 colonnes) : x = 0,5 ou 1,0  ; y = 0,5 × rangee + 0,25
//   tape horizontal (au-dessus case) : x = 0,5 × col + 0,25 ; y = 0,5 × (rangee + 1)

position_tape := [ [0.5, 0.75], [1.0, 0.75],
                   [0.5, 1.75], [1.0, 1.75],
                   [0.5, 2.75], [1.0, 2.75],
                   [0.5, 3.75], [1.0, 3.75],
                   [0.5, 4.75], [1.0, 4.75] ]


FONCTION centreCase(case)                 // case [rangee, colonne] → metres [x, y]
    RETOURNER [0,5 × case[1] + 0,25 ,  0,5 × case[0] + 0,25]


FONCTION prochaineCase(case, cote)
    SI cote = HAUT    RETOURNER [case[0] + 1, case[1]]
    SI cote = BAS     RETOURNER [case[0] − 1, case[1]]
    SI cote = DROITE  RETOURNER [case[0], case[1] + 1]
    SI cote = GAUCHE  RETOURNER [case[0], case[1] − 1]


//CHANGERRRRR
FONCTION detection_tape
  POUR i:= 0 a taille du tableau position_tape - 1
   SI pos_robot[1] est egale a position_tape[i][1] 
    SI pos_robot[0] < position_tape[i][0] < prochaine_pos[0]
     RETOURNE 1
    SI pos_robot[0] > position_tape[i][0] > prochaine_pos[0]
     RETOURNE 1
   SI pos_robot[0] est egale a position_tape[i][0]  
    SI pos_robot[1] < position_tape[i][1] < prochaine_pos[1] 
     RETOURNE 1
    SI pos_robot[1] > position_tape[i][1] > prochaine_pos[1]
     RETOURNE 1
  RETOURNE 0


FONCTION placerTapes()                   
    POUR chaque case (r, c) de la grille
        POUR chaque cote
            voisin ← prochaineCase([r, c], cote)
            SI voisin est dans la grille ET detectionTape([r, c], voisin)
                ajouterMur([r, c], cote)


// ===================== 2. CARTE =====================

FONCTION ajouterMur(case, cote)
    mur[case][cote] ← VRAI
    connu[case][cote] ← VRAI
    voisin ← prochaineCase(case, cote)
    SI voisin est dans la grille
        mur[voisin][cote oppose] ← VRAI
        connu[voisin][cote oppose] ← VRAI


FONCTION marquerLibre(case, cote)
    connu[case][cote] ← VRAI
    voisin ← prochaineCase(case, cote)
    SI voisin est dans la grille
        connu[voisin][cote oppose] ← VRAI


FONCTION initCarte()
    tout mettre a FAUX dans mur et connu
    POUR chaque rangee r
        ajouterMur([r, 0], GAUCHE)
        ajouterMur([r, 2], DROITE)
    POUR chaque colonne c
        ajouterMur([0, c], BAS)
        ajouterMur([9, c], HAUT)
    placerTapes()


// ===================== 3. DEPLACEMENT =====================

FONCTION arret()
    moteur gauche ← 0, moteur droit ← 0

FONCTION avance()
    moteur gauche ← VITESSE, moteur droit ← VITESSE

FONCTION recule()
    moteur gauche ← −0,5 × VITESSE, moteur droit ← −0,5 × VITESSE

FONCTION tourneDroit()
    moteur gauche ← +0,5 × VITESSE, moteur droit ← −0,5 × VITESSE

FONCTION tourneGauche()
    moteur gauche ← −0,5 × VITESSE, moteur droit ← +0,5 × VITESSE


FONCTION avancerCase()
    cible ← TAILLE_CASE convertie en pulses
    encodeurs ← 0
    somme ← 0

    TANT QUE moyenne(encG, encD) < cible
        erreur ← encG − encD
        somme  ← somme + erreur
        limiter somme entre −MAX_SOMME et +MAX_SOMME

        correction ← KP × erreur + KI × somme

        moteur gauche ← VITESSE − correction (MAX 1 et -1)
        moteur droit  ← VITESSE + correction (MAX 1 et -1)
        attendre 10 ms

    arret(), attendre 200 ms


FONCTION tourner(quarts)                  // +1 droite, −1 gauche, 2 demi-tour
    cible ← (π × ENTRAXE / 4) × |quarts|, convertie en pulses
    encodeurs ← 0
    TANT QUE moyenne(|encG|, |encD|) < cible
        SI quarts > 0  tourneDroit()
        SINON          tourneGauche()
    arret(), attendre 200 ms
    direction ← (direction + quarts) modulo 4


FONCTION faireFace(coteVoulu)
    diff ← (coteVoulu − direction) modulo 4
    SI diff = 1  tourner(+1)
    SI diff = 2  tourner(2)
    SI diff = 3  tourner(−1)


// ===================== 4. DETECTION D'OBSTACLE =====================

FONCTION murDevant()
    obstacles ← 0
    REPETER 7 fois
        vert  ← lire PIN_VERT
        rouge ← lire PIN_ROUGE
        SI vert = LOW ET rouge = LOW
            obstacles ← obstacles + 1
        attendre 3 ms
    RETOURNER obstacles ≥ 4


FONCTION scannerCellule()
    POUR cote DANS {devant, gauche, droite, derriere}
        SI connu[position_robot][cote] = FAUX
            faireFace(cote)
            SI murDevant()
                ajouterMur(position_robot, cote)
            SINON
                marquerLibre(position_robot, cote)


FONCTION calculerDistances(cibles)
    toutes les distances ← INFINI
    POUR chaque cible
        distance[cible] ← 0
        ajouter la cible a la file
    TANT QUE la file n'est pas vide
        case ← retirer la premiere case de la file
        POUR chaque cote de case sans mur
            voisin ← prochaineCase(case, cote)
            SI distance[voisin] = INFINI
                distance[voisin] ← distance[case] + 1
                ajouter le voisin a la file


FONCTION naviguerVers(cibles)
    TANT QUE position_robot n'est pas une des cibles
        SI bumperArriere()
            arret()
            RETOURNER echec

        scannerCellule()
        calculerDistances(cibles)
        SI distance[position_robot] = INFINI
            RETOURNER echec

        meilleur ← aucun
        POUR cote DANS {devant, droite, gauche, derriere}
            voisin ← prochaineCase(position_robot, cote)
            SI pas de mur de ce cote
               ET distance[voisin] < distance[case voisine de meilleur]
                meilleur ← cote

        prochaine_pos ← prochaineCase(position_robot, meilleur)
        faireFace(meilleur)
        SI murDevant()
            ajouterMur(position_robot, meilleur)
            recommencer la boucle

        avancerCase()
        position_robot ← prochaine_pos
    RETOURNER reussi


// ===================== 5. SIFFLET ET SONS =====================

FONCTION sifflet5kHz()                    // a revoir selon le filtre
    compteur ← 0
    REPETER 20 fois                       // environ 200 ms
        SI (lire PIN_5KHZ − lire PIN_AMBIANT) > SEUIL_SIFFLET
            compteur ← compteur + 1
        attendre 10 ms
    RETOURNER compteur ≥ 18

FONCTION bumperArriere()
    RETOURNER etat du bumper arriere (3)

FONCTION beep(nombre)
    REPETER nombre fois
        buzzer ON, attendre 100 ms
        buzzer OFF, attendre 100 ms
    attendre 400 ms


// ===================== 6. PROGRAMME PRINCIPAL =====================

FONCTION setup()
    BoardInit()
    PIN_VERT et PIN_ROUGE en ENTREE
    initCarte()
    position_robot ← [0, 1]
    direction ← HAUT
    etat ← ATTENTE
    beep(3)                       


FONCTION loop()
    SELON etat
        ATTENTE :
            SI sifflet5kHz() OU bumperArriere()
                beep(2), attendre 500 ms
                etat ← ALLER
        ALLER :
            SI naviguerVers(FIN_CIBLES)
                beep(1), etat ← RETOUR
            SINON
                etat ← ERREUR
        RETOUR :
            SI naviguerVers(DEPART_CIBLES)
                faireFace(HAUT), beep(3), etat ← TERMINE
            SINON
                etat ← ERREUR
        ERREUR :
            arret(), beep(5), etat ← TERMINE
        TERMINE :
            arret()

FIN
